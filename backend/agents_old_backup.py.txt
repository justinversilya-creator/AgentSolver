"""
Research Agent: retrieves real search results through a configured provider,
groups them into structured sources, and returns a validated ResearchResult.

The agent does not use an LLM's pretrained knowledge as evidence. Every source
comes from the search provider. If no provider is configured, the agent returns
an explicit limitation instead of fabricated results.
"""

from __future__ import annotations

import logging
import re
from dataclasses import dataclass, field
from typing import Protocol

from backend.schemas.evidence import (
    Confidence,
    ResearchFinding,
    ResearchResult,
    ResearchSource,
    RetrievalStatus,
)

logger = logging.getLogger("agentsolve.research")

MAX_SOURCES = 25
MAX_EXCERPT_CHARS = 2000
_STOPWORDS = {
    "a", "an", "and", "are", "as", "at", "be", "by", "compared", "do", "does", "for", "from",
    "how", "in", "is", "it", "of", "on", "or", "that", "the", "their", "to", "vs", "was",
    "what", "when", "which", "who", "why", "with", "than", "more", "less", "improve", "improves",
}


@dataclass(frozen=True)
class SearchRecord:
    """One record returned by a search or retrieval provider. Fields are optional and never invented."""

    title: str | None
    url: str | None = None
    publisher: str | None = None
    publication_date: str | None = None
    authors: list[str] = field(default_factory=list)
    doi: str | None = None
    abstract: str | None = None
    content_retrieved: bool = False
    retrieval_failed: bool = False


class SearchProvider(Protocol):
    """Interface any search provider must implement. Implementations must be asynchronous."""

    name: str

    async def search(self, query: str, max_results: int) -> list[SearchRecord]:
        """Return up to max_results records. Raise ProviderError on failure."""
        ...


class ProviderError(Exception):
    """Raised by a provider when a search call fails."""


class ProviderNotConfiguredError(Exception):
    """Raised when no search provider is configured."""


def split_subquestions(question: str) -> list[str]:
    """
    Derive simple research areas from the question's keywords.
    This is deterministic text processing, not a factual claim about the topic.
    """
    tokens = re.findall(r"[A-Za-z][A-Za-z\-]{2,}", question.lower())
    terms = [t for t in tokens if t not in _STOPWORDS]
    unique: list[str] = []
    for term in terms:
        if term not in unique:
            unique.append(term)
    return unique[:6]


def build_query(question: str) -> str:
    """Use the cleaned question as the primary query. Keep it short and literal."""
    return " ".join(question.split())[:300]


def _excerpt(text: str | None) -> str | None:
    if not text or not text.strip():
        return None
    return text.strip()[:MAX_EXCERPT_CHARS]


def _to_source(index: int, record: SearchRecord) -> ResearchSource:
    """Convert a provider record to a ResearchSource. Only returned fields are kept."""
    if record.retrieval_failed:
        status = RetrievalStatus.FAILED
    elif record.content_retrieved and record.abstract:
        status = RetrievalStatus.CONTENT_RETRIEVED
    else:
        status = RetrievalStatus.METADATA_ONLY

    limitations: list[str] = []
    if status is RetrievalStatus.METADATA_ONLY:
        limitations.append("Only search-result metadata was returned. The source content was not inspected.")
    if status is RetrievalStatus.FAILED:
        limitations.append("Content retrieval failed. Only metadata may be available.")
    if not record.url:
        limitations.append("No URL was returned for this source.")

    return ResearchSource(
        source_id=f"S{index}",
        title=(record.title or "Untitled record").strip()[:500],
        url=record.url,
        publisher=record.publisher,
        publication_date=record.publication_date,
        authors=[a for a in record.authors if a and a.strip()][:50],
        doi=record.doi,
        retrieved_excerpt=_excerpt(record.abstract) if status is RetrievalStatus.CONTENT_RETRIEVED else None,
        retrieval_status=status,
        metadata_returned=True,
        content_retrieved=status is RetrievalStatus.CONTENT_RETRIEVED,
        claims_supported=[],
        limitations=limitations,
    )


def _findings_from_sources(sources: list[ResearchSource]) -> list[ResearchFinding]:
    """
    Create findings only from returned abstracts. Each finding is the source's own
    abstract text, so the claim is traceable and not generated by the agent.
    """
    findings: list[ResearchFinding] = []
    for source in sources:
        if source.retrieval_status is not RetrievalStatus.CONTENT_RETRIEVED or not source.retrieved_excerpt:
            continue
        index = len(findings) + 1
        findings.append(
            ResearchFinding(
                finding_id=f"F{index}",
                claim=f"Abstract of {source.source_id} states: {source.retrieved_excerpt[:400]}",
                source_ids=[source.source_id],
                confidence=Confidence.LOW,
                limitations=[
                    "Based on the abstract only, not full text.",
                    "Confidence is not proof that the claim is correct.",
                ],
            )
        )
    return findings


def _build_limitations(
    sources: list[ResearchSource],
    subquestions: list[str],
    provider_errors: list[str],
) -> tuple[list[str], list[str], list[str]]:
    gaps: list[str] = []
    conflicts: list[str] = []
    limits: list[str] = []

    if not sources:
        gaps.append("The search provider returned no records for this question.")
    metadata_only = [s.source_id for s in sources if s.retrieval_status is RetrievalStatus.METADATA_ONLY]
    if metadata_only:
        gaps.append(f"Content was not inspected for: {', '.join(metadata_only)}.")
    if not any(s.content_retrieved for s in sources):
        gaps.append("No source content was retrieved, so no finding is based on inspected text.")

    limits.append("Search relevance was not evaluated by a human reviewer.")
    limits.append("Only the first search query was used. Subquestions were not searched separately.")
    if subquestions:
        limits.append(f"Research areas identified from keywords: {', '.join(subquestions)}.")
    for error in provider_errors:
        limits.append(f"Search provider error: {error}")

    if len(sources) >= 2:
        conflicts.append(
            "Conflicting findings were not assessed automatically. Compare the sources' abstracts manually."
        )

    return gaps, conflicts, limits


class ResearchAgent:
    """Runs the search provider and assembles a validated ResearchResult."""

    def __init__(self, provider: SearchProvider | None, max_results: int = 10) -> None:
        self._provider = provider
        self._max_results = max(1, min(max_results, MAX_SOURCES))

    async def run(self, question: str, request_id: str) -> ResearchResult:
        """
        Search for the question and return a ResearchResult.
        Raises ProviderNotConfiguredError if no provider is set, so callers can report the configuration gap.
        """
        if self._provider is None:
            raise ProviderNotConfiguredError(
                "No search provider is configured. Set SEARCH_PROVIDER and its API settings."
            )

        query = build_query(question)
        subquestions = split_subquestions(question)
        provider_errors: list[str] = []

        try:
            records = await self._provider.search(query, self._max_results)
        except ProviderError as exc:
            logger.warning("Search provider failed (request_id=%s): %s", request_id, exc)
            provider_errors.append("The search provider request failed.")
            records = []

        sources = [_to_source(i, r) for i, r in enumerate(records[:MAX_SOURCES], start=1)]
        findings = _findings_from_sources(sources)
        gaps, conflicts, limits = _build_limitations(sources, subquestions, provider_errors)

        return ResearchResult(
            agent="research",
            question=question,
            key_findings=findings,
            sources=sources,
            evidence_gaps=gaps,
            conflicting_findings=conflicts,
            research_limitations=limits,
        )
